Build an R/Bioconductor POC: Disease-Specific Network “Cliques” from BioGRID + Transcriptomics

You are Cursor. Generate a complete R package repo that identifies disease-relevant small dense subgraphs (“cliques”) by overlaying transcriptomics on a BioGRID regulatory/genetic scaffold. Provide working code, unit tests, a targets pipeline, toy data, and a minimal run path. Bake in reproducibility (renv), Parquet IO, provenance, and baselines-before-bells.

Objectives

Ingest BioGRID regulatory/genetic edges (pinned release) and transcriptomics (healthy vs disease).

Build condition-specific weighted networks (edge weights from coexpression Fisher-z; option for partial correlation later).

Score differential edges Δz = z_disease − z_healthy with uncertainty + BH FDR.

Discover small dense subgraphs (k-cliques/trusses) on the disease network; rank by Δz, stability, and pathway coherence.

Emit a Markdown decision card with top edges/cliques, enrichment, and provenance.

Follow: replicate at the correct unit, pseudobulk-by-donor for scRNA, explicit design & contrasts, stable containers/env, and evidence with uncertainty.

load_expression <- function(path){
  dt <- as.data.table(arrow::read_parquet(path))
  # Expect columns: gene + one column per SampleID? If long, pivot before.
  if (!"gene" %in% names(dt)) stop("counts parquet must contain 'gene'")
  setkey(dt, gene)
  return(dt)
}

load_metadata <- function(path){
  dt <- as.data.table(arrow::read_parquet(path))
  stopifnot(all(c("SampleID","SubjectID","Condition") %in% names(dt)))
  setkey(dt, SampleID)
  dt
}

  invisible(path)
}

ingest_biogrid <- function(tsv_path, release, keep_types=c("regulatory","genetic")){
  dt <- data.table::fread(tsv_path, sep="\t")

  dt[, `:=`(release = release)]
  unique(dt[, .(A,B,edge_type,evidence,pmid,release)])
}

R/weights.R

fisher_z <- function(r) atanh(pmax(pmin(r, 0.999999), -0.999999))

# Compute per-condition Fisher-z coexpression matrices using Pearson/Spearman
condition_fisher_z <- function(expr_dt, meta_dt, condition_col="Condition", method="pearson", fisher=TRUE){
  # expr_dt: gene x samples (wide). Convert to matrix with rownames=gene.
  genes <- expr_dt$gene
  mat <- as.matrix(expr_dt[, -c("gene")])
  rownames(mat) <- genes
  conds <- unique(meta_dt[[condition_col]])
  z_list <- list()
  for (cond in conds) {
    cols <- meta_dt[Condition == cond, SampleID]
    cols <- intersect(colnames(mat), cols)
    stopifnot(length(cols) >= 4)
    sub <- mat[, cols, drop=FALSE]
    if (method == "spearman") {
      r <- suppressWarnings(cor(t(apply(sub, 1, rank)), method="pearson"))
    } else {
      r <- suppressWarnings(cor(t(sub), method="pearson"))
    }
    z <- if (fisher) fisher_z(r) else r
    z_list[[cond]] <- z
  }
  z_list
}


R/diff_edges.R

score_diff_edges <- function(edges_dt, z_by_cond, meta_dt, healthy_label="healthy", disease_label="disease", fdr=0.05){
  stopifnot(healthy_label %in% names(z_by_cond), disease_label %in% names(z_by_cond))
  z_h <- z_by_cond[[healthy_label]]
  z_d <- z_by_cond[[disease_label]]
  # Map genes to indices
  genes <- rownames(z_h)
  gi <- function(g) match(g, genes)
  idxA <- gi(edges_dt$A); idxB <- gi(edges_dt$B)
  valid <- which(!is.na(idxA) & !is.na(idxB))
  dz <- rep(NA_real_, nrow(edges_dt))
  p  <- rep(1, nrow(edges_dt))
  # Large-sample variance for Fisher-z correlation difference (approx)
  se <- sqrt(2/(ncol(z_h)-3))
  for (k in valid){
    i <- idxA[k]; j <- idxB[k]
    dz[k] <- z_d[i,j] - z_h[i,j]
    zstat <- dz[k] / se
    p[k] <- 2*pnorm(-abs(zstat))
  }
  out <- data.table::as.data.table(edges_dt)
  out[, `:=`(delta_z = dz, pval = p)]
  out[, fdr := p.adjust(pval, method="BH")]
  out[order(-delta_z)]
}


R/cliques.R

top_edges_graph <- function(edges_scored, pct=5){
  n <- max(1, floor(nrow(edges_scored) * pct/100))
  sub <- edges_scored[order(-delta_z)][1:n]
  igraph::graph_from_data_frame(sub[, .(A,B,delta_z)], directed=FALSE) |>
    igraph::simplify()
}

rank_cliques <- function(G, ks=c(3,4)){
  cl <- igraph::max_cliques(G)
  keep <- Filter(function(x) length(x) %in% ks, cl)
  if (length(keep)==0) return(data.table::data.table(clique=character(), score=numeric()))
  score_one <- function(nodes){
    # mean edge delta_z over clique edges
    eds <- combn(nodes, 2, simplify=FALSE)
    vals <- vapply(eds, function(e) igraph::edge_attr(G, "delta_z", index=igraph::E(G, P = e)), numeric(1))
    mean(vals, na.rm=TRUE)
  }
  rows <- lapply(keep, function(v) data.table::data.table(clique=paste(sort(igraph::V(G)[v]$name), collapse="|"),
                                                          score=score_one(igraph::V(G)[v]$name)))
  data.table::rbindlist(rows)[order(-score)]
}


R/enrichment.R

clique_enrichment <- function(cliques_tbl, gene_sets_parquet){
  if (nrow(cliques_tbl)==0) return(data.table::data.table())
  gs <- as.data.table(arrow::read_parquet(gene_sets_parquet)) # expect columns: set, gene
  universe <- unique(gs$gene)
  top_genes <- unique(unlist(strsplit(cliques_tbl$clique[1:min(50, nrow(cliques_tbl))], "\\|")))
  # Hypergeometric test per set
  library(stats)
  enr <- gs[, .(K = .N, k = sum(gene %in% top_genes)), by=set]
  N <- length(universe); n <- length(top_genes)
  enr[, p := phyper(q=k-1, m=K, n=N-K, k=n, lower.tail=FALSE)]
  enr[, fdr := p.adjust(p, method="BH")]
  enr[order(p)]
}


R/report.R

render_decision_card <- function(run_id, edges, cliques, enrich, release_dir){
  dir.create(release_dir, showWarnings=FALSE, recursive=TRUE)
  path <- file.path(release_dir, sprintf("decision_card_%s.md", run_id))
  head_edges <- utils::head(edges[, .(A,B,delta_z,fdr)][order(fdr)], 20)
  head_cliq  <- utils::head(cliques, 20)
  head_enr   <- utils::head(enrich, 20)
  md <- paste0(
    "# Network Clicks POC — Decision Card (", run_id, ")\n\n",
    "## Top Edges (Δz, FDR)\n\n", knitr::kable(head_edges, format="markdown"), "\n\n",
    "## Top Cliques (mean Δz)\n\n", knitr::kable(head_cliq, format="markdown"), "\n\n",
    "## Enrichment\n\n", knitr::kable(head_enr, format="markdown"), "\n\n",
    "## Provenance\n\n```\n", capture.output(utils::sessionInfo()), "\n```"
  )
  writeLines(md, path)
  path
}
